# ============================================================
# PRIMARY FIX, try this first.
#
# Add this block to the .htaccess file in the SAME DIRECTORY as
# bestbrainplus.com's live index.html on Hostinger (open it via
# hPanel's File Manager, or FTP). If a .htaccess already exists
# there, paste this block into it, do not overwrite the file.
#
# Why this is needed: the app's own vercel.json already contains
# this exact rewrite (backend-api/* -> the EC2 backend), but that
# file only means something to Vercel. Hostinger has never seen
# it, so every login/signup/API call was hitting Hostinger's own
# server with nothing behind that path, which is the "server is
# temporarily busy" 503 you were seeing.
# ============================================================

RewriteEngine On

# Forward API calls to the real backend (EC2), mirroring vercel.json.
RewriteCond %{REQUEST_URI} ^/backend-api/
RewriteRule ^backend-api/(.*)$ https://api.bestbrainplus.com/$1 [P,L]

# Same for the realtime/live-class connection.
RewriteCond %{REQUEST_URI} ^/socket.io/
RewriteRule ^socket\.io/(.*)$ https://api.bestbrainplus.com/socket.io/$1 [P,L]
